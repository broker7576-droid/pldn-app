        void fetchApk(String url, java.io.File f, java.io.File marker) throws Exception {
            long have = 0;
            String prev = "";
            try {
                if (marker.exists()) {
                    java.io.BufferedReader br = new java.io.BufferedReader(new java.io.FileReader(marker));
                    prev = br.readLine();
                    br.close();
                }
            } catch (Exception e) { prev = ""; }
            if (f.exists() && url.equals(prev)) { have = f.length(); } else if (f.exists()) { f.delete(); }
            java.io.FileWriter fw = new java.io.FileWriter(marker);
            fw.write(url);
            fw.close();
            java.net.HttpURLConnection c = (java.net.HttpURLConnection) new java.net.URL(url).openConnection();
            c.setConnectTimeout(15000);
            c.setReadTimeout(30000);
            c.setInstanceFollowRedirects(true);
            if (have > 0) { c.setRequestProperty("Range", "bytes=" + have + "-"); }
            int code = c.getResponseCode();
            if (code == 416) { f.delete(); throw new Exception("HTTP 416"); }
            if (code != 200 && code != 206) { throw new Exception("HTTP " + code); }
            boolean resumed = (code == 206 && have > 0);
            long got = resumed ? have : 0;
            long total = c.getContentLengthLong();
            if (total > 0) { total = total + got; }
            java.io.InputStream in = c.getInputStream();
            java.io.FileOutputStream out = new java.io.FileOutputStream(f, resumed);
            byte[] buf = new byte[65536];
            int n;
            try {
                while ((n = in.read(buf)) > 0) {
                    out.write(buf, 0, n);
                    got += n;
                    bytes = got;
                    if (total > 0) { pct = (int) (got * 100 / total); }
                }
            } finally {
                try { out.close(); } catch (Exception e) { }
                try { in.close(); } catch (Exception e) { }
            }
            if (total > 0 && got < total) { throw new Exception("uzilib qoldi"); }
            if (got < 1000000) { f.delete(); throw new Exception("fayl kichik: " + got); }
        }
        @android.webkit.JavascriptInterface

            if ("downloading".equals(state)) { return; }
            state = "downloading"; pct = 0; bytes = 0; err = "";
            new Thread(new Runnable() {

                    java.io.File f = new java.io.File(getCacheDir(), "pldn-update.apk");
                    java.io.File marker = new java.io.File(getCacheDir(), "pldn-update.url");
                    int tries = 0;

                        }
                    }
                }
            }).start();
        }

        java.io.File vin;
        java.io.File vout;
        java.io.OutputStream vos;
        volatile String state = "idle";
        volatile int pct = 0;
        volatile String err = "";
        @android.webkit.JavascriptInterface

            try {

                vin = new java.io.File(getCacheDir(), "pldn-vin.bin");
                vout = new java.io.File(getCacheDir(), "pldn-vout.mp4");
                vos = new java.io.FileOutputStream(vin);

            } catch (Exception e) { return false; }

        @android.webkit.JavascriptInterface

        @android.webkit.JavascriptInterface
        public void trim(final double startSec, final double lenSec) {
            state = "working"; pct = 0; err = "";
            new Thread(new Runnable() {
                public void run() {
                    android.media.MediaExtractor ex = null;
                    android.media.MediaMuxer mx = null;
                    boolean started = false;
                    try {
                        if (vout.exists()) { vout.delete(); }
                        ex = new android.media.MediaExtractor();
                        ex.setDataSource(vin.getAbsolutePath());
                        int vt = -1;
                        for (int i = 0; i < ex.getTrackCount(); i++) {
                            String m = ex.getTrackFormat(i).getString(android.media.MediaFormat.KEY_MIME);
                            if (m != null && m.startsWith("video/")) { vt = i; break; }
                        }
                        if (vt < 0) { throw new Exception("video trek yoq"); }
                        ex.selectTrack(vt);
                        android.media.MediaFormat fmt = ex.getTrackFormat(vt);
                        int rot = 0;
                        try {
                            android.media.MediaMetadataRetriever r = new android.media.MediaMetadataRetriever();
                            r.setDataSource(vin.getAbsolutePath());
                            String rs = r.extractMetadata(android.media.MediaMetadataRetriever.METADATA_KEY_VIDEO_ROTATION);
                            if (rs != null) { rot = Integer.parseInt(rs); }
                            r.release();
                        } catch (Exception e) { rot = 0; }
                        mx = new android.media.MediaMuxer(vout.getAbsolutePath(), android.media.MediaMuxer.OutputFormat.MUXER_OUTPUT_MPEG_4);
                        mx.setOrientationHint(rot);
                        int ot = mx.addTrack(fmt);
                        mx.start();
                        started = true;
                        long lenUs = (long) (lenSec * 1000000.0);
                        ex.seekTo((long) (startSec * 1000000.0), android.media.MediaExtractor.SEEK_TO_PREVIOUS_SYNC);
                        int cap = 8 * 1024 * 1024;
                        if (fmt.containsKey(android.media.MediaFormat.KEY_MAX_INPUT_SIZE)) { cap = Math.max(cap, fmt.getInteger(android.media.MediaFormat.KEY_MAX_INPUT_SIZE)); }
                        java.nio.ByteBuffer buf = java.nio.ByteBuffer.allocate(cap);
                        android.media.MediaCodec.BufferInfo bi = new android.media.MediaCodec.BufferInfo();
                        long base = -1;
                        while (true) {
                            buf.clear();
                            int n = ex.readSampleData(buf, 0);
                            if (n < 0) { break; }
                            long t = ex.getSampleTime();
                            if (base < 0) { base = t; }
                            if (t - base >= lenUs) { break; }
                            bi.offset = 0;
                            bi.size = n;
                            bi.presentationTimeUs = t - base;
                            bi.flags = ex.getSampleFlags();
                            mx.writeSampleData(ot, buf, bi);
                            pct = (int) ((t - base) * 100 / lenUs);
                            ex.advance();
                        }
                        mx.stop();
                        started = false;
                        pct = 100;
                        state = "done";
                    } catch (Throwable e) {
                        err = String.valueOf(e);
                        state = "error";
                    } finally {
                        try { if (mx != null) { if (started) { mx.stop(); } mx.release(); } } catch (Throwable e) { }
                        try { if (ex != null) { ex.release(); } } catch (Throwable e) { }
                    }
                }
            }).start();
        }

            return state + "|" + pct + "|" + err;

        }

    public class PldnVoice {
        android.speech.SpeechRecognizer sr;
        android.speech.tts.TextToSpeech tts;
        volatile boolean ttsReady = false;
        volatile String pendText = "";
        volatile String pendLang = "uz-UZ";
        void emit(final org.json.JSONObject o) {
            runOnUiThread(new Runnable() {
                public void run() {
                    try {
                        getBridge().getWebView().evaluateJavascript("window.pldnVoiceEvent&&window.pldnVoiceEvent(" + o.toString() + ")", null);
                    } catch (Throwable e) { }
                }
            });
        }
        void emitType(String type, String key, Object val) {
            try {
                org.json.JSONObject o = new org.json.JSONObject();
                o.put("type", type);
                if (key != null) { o.put(key, val); }
                emit(o);
            } catch (Throwable e) { }
        }
        void killRec() {
            try {
                if (sr != null) {
                    sr.setRecognitionListener(null);
                    sr.cancel();
                    sr.destroy();
                    sr = null;
                }
            } catch (Throwable e) { sr = null; }
        }
        @android.webkit.JavascriptInterface
        public boolean available() {
            try { return android.speech.SpeechRecognizer.isRecognitionAvailable(MainActivity.this); } catch (Throwable e) { return false; }
        }
        @android.webkit.JavascriptInterface
        public boolean hasMic() {
            return ContextCompat.checkSelfPermission(MainActivity.this, Manifest.permission.RECORD_AUDIO) == PackageManager.PERMISSION_GRANTED;
        }
        @android.webkit.JavascriptInterface
        public void askMic() {
            runOnUiThread(new Runnable() {
                public void run() {
                    ActivityCompat.requestPermissions(MainActivity.this, new String[]{Manifest.permission.RECORD_AUDIO}, 1);
                }
            });
        }
        @android.webkit.JavascriptInterface
        public void start(final String lang) {
            runOnUiThread(new Runnable() {
                public void run() {
                    try {
                        killRec();
                        sr = android.speech.SpeechRecognizer.createSpeechRecognizer(MainActivity.this);
                        sr.setRecognitionListener(new android.speech.RecognitionListener() {
                            public void onReadyForSpeech(android.os.Bundle p) { emitType("ready", null, null); }
                            public void onBeginningOfSpeech() { emitType("begin", null, null); }
                            public void onRmsChanged(float rms) { }
                            public void onBufferReceived(byte[] b) { }
                            public void onEndOfSpeech() { emitType("end", null, null); }
                            public void onError(int code) { emitType("error", "code", code); }
                            public void onResults(android.os.Bundle b) {
                                try {
                                    java.util.ArrayList<String> l = b.getStringArrayList(android.speech.SpeechRecognizer.RESULTS_RECOGNITION);
                                    org.json.JSONArray a = new org.json.JSONArray();
                                    if (l != null) { for (String s : l) { a.put(s); } }
                                    emitType("result", "texts", a);
                                } catch (Throwable e) { emitType("error", "code", -1); }
                            }
                            public void onPartialResults(android.os.Bundle b) {
                                try {
                                    java.util.ArrayList<String> l = b.getStringArrayList(android.speech.SpeechRecognizer.RESULTS_RECOGNITION);
                                    if (l != null && l.size() > 0) { emitType("partial", "text", l.get(0)); }
                                } catch (Throwable e) { }
                            }
                            public void onEvent(int t, android.os.Bundle b) { }
                        });
                        android.content.Intent i = new android.content.Intent(android.speech.RecognizerIntent.ACTION_RECOGNIZE_SPEECH);
                        i.putExtra(android.speech.RecognizerIntent.EXTRA_LANGUAGE_MODEL, android.speech.RecognizerIntent.LANGUAGE_MODEL_FREE_FORM);
                        i.putExtra(android.speech.RecognizerIntent.EXTRA_LANGUAGE, lang);
                        i.putExtra(android.speech.RecognizerIntent.EXTRA_PARTIAL_RESULTS, true);
                        i.putExtra(android.speech.RecognizerIntent.EXTRA_MAX_RESULTS, 4);
                        i.putExtra(android.speech.RecognizerIntent.EXTRA_CALLING_PACKAGE, getPackageName());
                        sr.startListening(i);
                    } catch (Throwable e) {
                        emitType("error", "code", -2);
                    }
                }
            });
        }
        @android.webkit.JavascriptInterface
        public void stop() {
            runOnUiThread(new Runnable() {
                public void run() { try { if (sr != null) { sr.stopListening(); } } catch (Throwable e) { } }
            });
        }
        @android.webkit.JavascriptInterface
        public void cancel() {
            runOnUiThread(new Runnable() {
                public void run() { killRec(); }
            });
        }
        void doSpeak(String text, String lang) {
            try {
                java.util.Locale loc = lang.startsWith("ru") ? new java.util.Locale("ru", "RU") : new java.util.Locale("uz", "UZ");
                int r = tts.isLanguageAvailable(loc);
                if (r < android.speech.tts.TextToSpeech.LANG_AVAILABLE) { emitType("tts", "ok", false); return; }
                tts.setLanguage(loc);
                tts.speak(text, android.speech.tts.TextToSpeech.QUEUE_FLUSH, null, "pldn");
                emitType("tts", "ok", true);
            } catch (Throwable e) { emitType("tts", "ok", false); }
        }
        @android.webkit.JavascriptInterface
        public void speak(final String text, final String lang) {
            runOnUiThread(new Runnable() {
                public void run() {
                    try {
                        if (tts == null) {
                            pendText = text; pendLang = lang;
                            tts = new android.speech.tts.TextToSpeech(MainActivity.this, new android.speech.tts.TextToSpeech.OnInitListener() {
                                public void onInit(int st) {
                                    ttsReady = (st == android.speech.tts.TextToSpeech.SUCCESS);
                                    if (ttsReady) { doSpeak(pendText, pendLang); } else { emitType("tts", "ok", false); }
                                }
                            });
                        } else if (ttsReady) {
                            doSpeak(text, lang);
                        }
                    } catch (Throwable e) { emitType("tts", "ok", false); }
                }
            });
        }
        @android.webkit.JavascriptInterface
        public void shutUp() {
            runOnUiThread(new Runnable() {
                public void run() { try { if (tts != null) { tts.stop(); } } catch (Throwable e) { } }
            });
        }
        void release() {
            killRec();
            try { if (tts != null) { tts.stop(); tts.shutdown(); tts = null; } } catch (Throwable e) { }
        }
    }
