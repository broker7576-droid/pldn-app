        void fetchApk(String url, java.io.File f, java.io.File marker) throws Exception {
            long have = 0;
            String prev = "";
            try {
                if (marker.exists()) {
                    java.io.BufferedReader br = new java.io.BufferedReader(new java.io.FileReader(marker));
                    prev = br.readLine();
                    br.close();
                }
            } catch (Exception e) { prev = ""; }
            if (f.exists() && url.equals(prev)) { have = f.length(); } else if (f.exists()) { f.delete(); }
            java.io.FileWriter fw = new java.io.FileWriter(marker);
            fw.write(url);
            fw.close();
            java.net.HttpURLConnection c = (java.net.HttpURLConnection) new java.net.URL(url).openConnection();
            c.setConnectTimeout(15000);
            c.setReadTimeout(30000);
            c.setInstanceFollowRedirects(true);
            if (have > 0) { c.setRequestProperty("Range", "bytes=" + have + "-"); }
            int code = c.getResponseCode();
            if (code == 416) { f.delete(); throw new Exception("HTTP 416"); }
            if (code != 200 && code != 206) { throw new Exception("HTTP " + code); }
            boolean resumed = (code == 206 && have > 0);
            long got = resumed ? have : 0;
            long total = c.getContentLengthLong();
            if (total > 0) { total = total + got; }
            java.io.InputStream in = c.getInputStream();
            java.io.FileOutputStream out = new java.io.FileOutputStream(f, resumed);
            byte[] buf = new byte[65536];
            int n;
            try {
                while ((n = in.read(buf)) > 0) {
                    out.write(buf, 0, n);
                    got += n;
                    bytes = got;
                    if (total > 0) { pct = (int) (got * 100 / total); }
                }
            } finally {
                try { out.close(); } catch (Exception e) { }
                try { in.close(); } catch (Exception e) { }
            }

            state = "working"; pct = 0; err = "";

                public void run() {
                    android.media.MediaExtractor ex = null;
                    android.media.MediaMuxer mx = null;
                    boolean started = false;
                    try {
                        if (vout.exists()) { vout.delete(); }
                        ex = new android.media.MediaExtractor();
                        ex.setDataSource(vin.getAbsolutePath());
                        int vt = -1;
                        for (int i = 0; i < ex.getTrackCount(); i++) {
                            String m = ex.getTrackFormat(i).getString(android.media.MediaFormat.KEY_MIME);
                            if (m != null && m.startsWith("video/")) { vt = i; break; }
                        }
                        if (vt < 0) { throw new Exception("video trek yoq"); }
                        ex.selectTrack(vt);
                        android.media.MediaFormat fmt = ex.getTrackFormat(vt);
                        int rot = 0;
                        try {
                            android.media.MediaMetadataRetriever r = new android.media.MediaMetadataRetriever();
                            r.setDataSource(vin.getAbsolutePath());
                            String rs = r.extractMetadata(android.media.MediaMetadataRetriever.METADATA_KEY_VIDEO_ROTATION);
                            if (rs != null) { rot = Integer.parseInt(rs); }
                            r.release();
                        } catch (Exception e) { rot = 0; }
                        mx = new android.media.MediaMuxer(vout.getAbsolutePath(), android.media.MediaMuxer.OutputFormat.MUXER_OUTPUT_MPEG_4);
                        mx.setOrientationHint(rot);
                        int ot = mx.addTrack(fmt);
                        mx.start();
                        started = true;
                        long lenUs = (long) (lenSec * 1000000.0);
                        ex.seekTo((long) (startSec * 1000000.0), android.media.MediaExtractor.SEEK_TO_PREVIOUS_SYNC);
                        int cap = 8 * 1024 * 1024;
                        if (fmt.containsKey(android.media.MediaFormat.KEY_MAX_INPUT_SIZE)) { cap = Math.max(cap, fmt.getInteger(android.media.MediaFormat.KEY_MAX_INPUT_SIZE)); }
                        java.nio.ByteBuffer buf = java.nio.ByteBuffer.allocate(cap);
                        android.media.MediaCodec.BufferInfo bi = new android.media.MediaCodec.BufferInfo();
                        long base = -1;
                        while (true) {
                            buf.clear();
                            int n = ex.readSampleData(buf, 0);
                            if (n < 0) { break; }
                            long t = ex.getSampleTime();
                            if (base < 0) { base = t; }
                            if (t - base >= lenUs) { break; }
                            bi.offset = 0;
                            bi.size = n;
                            bi.presentationTimeUs = t - base;
                            bi.flags = ex.getSampleFlags();
                            mx.writeSampleData(ot, buf, bi);
                            pct = (int) ((t - base) * 100 / lenUs);
                            ex.advance();
                        }
                        mx.stop();
                        started = false;
                        pct = 100;
                        state = "done";
                    } catch (Throwable e) {
                        err = String.valueOf(e);
                        state = "error";
                    } finally {
                        try { if (mx != null) { if (started) { mx.stop(); } mx.release(); } } catch (Throwable e) { }
                        try { if (ex != null) { ex.release(); } } catch (Throwable e) { }
                    }
                }

            smsRcv = new android.content.BroadcastReceiver() {
                public void onReceive(android.content.Context ctx, android.content.Intent it) {
                    int rc = getResultCode();
                    if (rc != android.app.Activity.RESULT_OK) { smsFail = rc; }
                    smsLeft = smsLeft - 1;
                    if (smsLeft <= 0) {
                        try {
                            org.json.JSONObject o = new org.json.JSONObject();
                            o.put("type", "sms");
                            o.put("ok", smsFail == 0);
                            o.put("code", smsFail);
                            emitPhone(o);
                        } catch (Throwable e) { }
                    }
                }
            };
            ContextCompat.registerReceiver(MainActivity.this, smsRcv, new android.content.IntentFilter(SMS_ACT), ContextCompat.RECEIVER_NOT_EXPORTED);
            smsRegistered = true;
        }
        @android.webkit.JavascriptInterface
        public String sms(String number, String text) {
            if (!has("sms")) { return "noperm"; }
            try {
                String n = cleanNum(number);
                if (n.length() < 3) { return "err:raqam"; }
                if (text == null || text.trim().length() == 0) { return "err:matn"; }
                ensureSmsRcv();
                android.telephony.SmsManager sm;
                if (android.os.Build.VERSION.SDK_INT >= 31) { sm = getSystemService(android.telephony.SmsManager.class); }
                else { sm = android.telephony.SmsManager.getDefault(); }
                java.util.ArrayList<String> parts = sm.divideMessage(text);
                java.util.ArrayList<android.app.PendingIntent> sent = new java.util.ArrayList<android.app.PendingIntent>();
                smsLeft = parts.size();
                smsFail = 0;
                for (int k = 0; k < parts.size(); k++) {
                    android.content.Intent si = new android.content.Intent(SMS_ACT);
                    si.setPackage(getPackageName());
                    sent.add(android.app.PendingIntent.getBroadcast(MainActivity.this, 700 + k, si, android.app.PendingIntent.FLAG_IMMUTABLE | android.app.PendingIntent.FLAG_UPDATE_CURRENT));
                }
                if (parts.size() > 1) { sm.sendMultipartTextMessage(n, null, parts, sent, null); }
                else { sm.sendTextMessage(n, null, text, sent.get(0), null); }
                return "ok";
            } catch (Throwable e) { return "err:" + e; }
        }

        @android.webkit.JavascriptInterface
        public String apps() {
            try {
                android.content.pm.PackageManager pm = getPackageManager();
                android.content.Intent q = new android.content.Intent(android.content.Intent.ACTION_MAIN);
                q.addCategory(android.content.Intent.CATEGORY_LAUNCHER);
                java.util.List<android.content.pm.ResolveInfo> l = pm.queryIntentActivities(q, 0);
                java.util.HashSet<String> seen = new java.util.HashSet<String>();
                org.json.JSONArray a = new org.json.JSONArray();
                for (android.content.pm.ResolveInfo ri : l) {
                    String pkg = ri.activityInfo.packageName;
                    if (pkg.equals(getPackageName()) || seen.contains(pkg)) { continue; }
                    seen.add(pkg);
                    org.json.JSONObject o = new org.json.JSONObject();
                    o.put("n", String.valueOf(ri.loadLabel(pm)));
                    o.put("p", pkg);
                    a.put(o);
                }
                return a.toString();
            } catch (Throwable e) { return "err:" + e; }
        }

        @android.webkit.JavascriptInterface
        public String volume(String stream, String op, int pct) {
            try {
                android.media.AudioManager am = (android.media.AudioManager) getSystemService(android.content.Context.AUDIO_SERVICE);
                int st = android.media.AudioManager.STREAM_MUSIC;
                if ("ring".equals(stream)) { st = android.media.AudioManager.STREAM_RING; }
                else if ("alarm".equals(stream)) { st = android.media.AudioManager.STREAM_ALARM; }
                int max = am.getStreamMaxVolume(st);
                int cur = am.getStreamVolume(st);
                int step = Math.max(1, Math.round(max * (pct > 0 ? pct : 20) / 100f));
                int v = cur;
                if ("set".equals(op)) { v = Math.round(Math.max(0, Math.min(100, pct)) * max / 100f); }
                else if ("up".equals(op)) { v = Math.min(max, cur + step); }
                else if ("down".equals(op)) { v = Math.max(0, cur - step); }
                else if ("mute".equals(op)) { v = 0; }
                am.setStreamVolume(st, v, android.media.AudioManager.FLAG_SHOW_UI);
                return "ok|" + Math.round(v * 100f / max);
            } catch (Throwable e) { return "err:" + e; }
        }
